-- Portal Projetos | Visibilidade e diagnóstico do cronograma -- Execute no Supabase SQL Editor após as migrations de 31/08/2026. -- Incremental e segura para nova execução. begin; -- Visualizar Cronograma passa a conceder a leitura da lista de projetos -- necessária para selecionar e abrir um cronograma. Não concede criar, -- editar ou excluir projetos no Portfólio. create or replace function public.can_access_project(p_project_id uuid) returns boolean language sql stable security definer set search_path = public, pg_temp as $$ select public.is_admin() or public.has_module_action('portfolio','view') or public.has_module_action('cronograma','view'); $$; revoke all on function public.can_access_project(uuid) from public; grant execute on function public.can_access_project(uuid) to authenticated; drop policy if exists projects_select on public.projects; create policy projects_select on public.projects for select to authenticated using (public.can_access_project(id)); -- A política restritiva anterior exigia Portfólio mesmo quando o operador -- já possuía Visualizar Cronograma. A leitura agora aceita uma das duas. drop policy if exists crud_guard_projects_select on public.projects; create policy crud_guard_projects_select on public.projects as restrictive for select to authenticated using ( public.has_module_action('portfolio','view') or public.has_module_action('cronograma','view') ); -- Reafirma as políticas permissivas de leitura compartilhada do cronograma. drop policy if exists tasks_select on public.tasks; create policy tasks_select on public.tasks for select to authenticated using (public.has_module_action('cronograma','view') and public.can_access_project(project_id)); drop policy if exists project_phases_select on public.project_phases; create policy project_phases_select on public.project_phases for select to authenticated using (public.has_module_action('cronograma','view') and public.can_access_project(project_id)); drop policy if exists task_schedule_versions_select on public.task_schedule_versions; create policy task_schedule_versions_select on public.task_schedule_versions for select to authenticated using (public.has_module_action('cronograma','view') and public.can_access_project(project_id)); drop policy if exists task_dependencies_select on public.task_dependencies; create policy task_dependencies_select on public.task_dependencies for select to authenticated using ( public.has_module_action('cronograma','view') and exists ( select 1 from public.tasks t where t.id = task_id and public.can_access_project(t.project_id) ) ); -- Retorna projeto, atividades e dependências na mesma leitura. Isso evita uma -- tela parcialmente carregada quando uma das consultas falha ou chega depois. create or replace function public.get_schedule_snapshot(p_project_id uuid) returns jsonb language plpgsql stable security definer set search_path = public, pg_temp as $$ declare v_project jsonb; v_tasks jsonb; v_dependencies jsonb; begin if auth.uid() is null then raise exception 'AUTH_REQUIRED'; end if; if not public.has_module_action('cronograma','view') then raise exception 'SCHEDULE_VIEW_DENIED'; end if; if not public.can_access_project(p_project_id) then raise exception 'PROJECT_VIEW_DENIED'; end if; select to_jsonb(p) into v_project from public.projects p where p.id = p_project_id; if v_project is null then raise exception 'PROJECT_NOT_FOUND'; end if; select coalesce(jsonb_agg(to_jsonb(t) order by t.created_at, t.id), '[]'::jsonb) into v_tasks from public.tasks t where t.project_id = p_project_id; select coalesce(jsonb_agg(to_jsonb(d) order by d.task_id, d.depends_on_task_id), '[]'::jsonb) into v_dependencies from public.task_dependencies d join public.tasks t on t.id = d.task_id where t.project_id = p_project_id; return jsonb_build_object( 'project', v_project, 'tasks', v_tasks, 'dependencies', v_dependencies, 'task_count', jsonb_array_length(v_tasks), 'server_time', now(), 'source', 'database' ); end; $$; revoke all on function public.get_schedule_snapshot(uuid) from public; grant execute on function public.get_schedule_snapshot(uuid) to authenticated; -- Diagnóstico administrativo: mostra a regra efetiva de um usuário sem -- exigir que o administrador faça login com a conta da pessoa. create or replace function public.diagnose_schedule_access(p_user_id uuid) returns jsonb language plpgsql stable security definer set search_path = public, pg_temp as $$ declare v_role text; v_name text; v_active boolean := false; v_schedule boolean := false; v_portfolio boolean := false; v_modules jsonb := '[]'::jsonb; v_projects jsonb := '[]'::jsonb; begin if not public.is_admin() then raise exception 'ADMIN_REQUIRED'; end if; select p.role::text, p.full_name, p.active into v_role, v_name, v_active from public.profiles p where p.id = p_user_id; if v_role is null then raise exception 'PROFILE_NOT_FOUND'; end if; if v_active and v_role = 'admin' then v_schedule := true; v_portfolio := true; v_modules := to_jsonb(array['portfolio','tap','eap','raci','cronograma','statusreport','riscos','pontosfocais','encaminhamentos','on_demand','tep','licoes','lecoms','solicitacoes']); elsif v_active and v_role = 'operator' then select coalesce(jsonb_agg(u.module_key order by u.module_key) filter (where coalesce(u.can_view,false)), '[]'::jsonb) into v_modules from public.user_module_permissions u where u.user_id = p_user_id; v_schedule := exists(select 1 from public.user_module_permissions u where u.user_id=p_user_id and u.module_key='cronograma' and coalesce(u.can_view,false)); v_portfolio := exists(select 1 from public.user_module_permissions u where u.user_id=p_user_id and u.module_key='portfolio' and coalesce(u.can_view,false)); elsif v_active and v_role = 'collaborator' then v_modules := '["solicitacoes"]'::jsonb; end if; if v_schedule then select coalesce(jsonb_agg(jsonb_build_object('id',p.id,'name',p.name,'control_code',p.control_code) order by p.name), '[]'::jsonb) into v_projects from public.projects p; end if; return jsonb_build_object( 'user_id', p_user_id, 'full_name', v_name, 'role', v_role, 'active', v_active, 'can_view_schedule', v_schedule, 'can_view_portfolio', v_portfolio, 'modules', v_modules, 'projects', v_projects, 'project_count', jsonb_array_length(v_projects), 'rule', 'cronograma_view_implies_project_list' ); end; $$; revoke all on function public.diagnose_schedule_access(uuid) from public; grant execute on function public.diagnose_schedule_access(uuid) to authenticated; commit; -- Conferência opcional após executar: -- select public.diagnose_schedule_access(''::uuid);
